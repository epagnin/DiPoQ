\section{Mathematical and computational background}

\subsection{Definition of lattices}
The fundamental quantity in lattice-based cryptography is a lattice. For the purposes of this text, the following definition is sufficient. 
\begin{definition}[Lattice]
    A lattice $\LL$ is a discrete additive subgroup of $\RR^\nd$.
\end{definition} 
A lattice is finitely generated by a set of linearly independent basis vectors $\{ b_1, \ldots, b_k \}$, where each $b_i \in \RR^k$, often collected in a basis matrix $B$ such that
\[ \LL = \LL(B) = B \cdot \ZZ^k = \left\{ \sum_{i = 1}^{k} z_i b_i : z_i \in \ZZ\right\}. \]
The number of linearly independent basis vectors $k$ is the \textit{rank} of the lattice. We are only intested in lattices of full rank, where $k = \nd$. A characteristic property of a given lattice $\LL$ are it's minima, which relate to the geometry of the lattice. This means the lattice must be endowed with a norm $|| \cdot ||$. Unless otherwise specified, we work in the Euclidean norm.
\begin{definition}[Successive minima] % replace this with general def of minima
    The $i$th minimum $\lambda_i(\LL)$ is the radius of the smallest sphere centred in the origin containing $i$ linearly independent lattice vectors:
    \[ \lambda_i(\LL) = \inf\{ r : \dim (\mathrm{span} (\LL \cap \mathcal{B}(0,r))) \geq i)\}. \]
\end{definition}
In particular, the \textit{first minimum}, also known as the \textit{minimum distance} or simply \textit{minimum} is the length of the shortest nonzero lattice vector 
    \[ \lambda_1(\LL) = \min_{v \in \LL \setminus \{ 0 \} } || v ||. \]
Every lattice $\LL$ induces a dual lattice defined as follows.
\begin{definition}[Dual lattice]
    The dual $\LL^*$ of a lattice $\LL$ is defined as
    \[ \LL^* = \{ u : \langle u, \LL \rangle \subseteq \ZZ \}. \]
    In words, the dual $\LL^*$ is the set of points having integer inner product with vectors in $\LL$. 
\end{definition}
The dual $\LL^*$ is also a lattice. If $B$ is a basis $\LL$, when $(B^T)^{-1}$ is a basis of $\LL^*$ ($T$ denotes transpose).

\subsection{Basic computational problems}

Cover SVP, BDD, GapSVP, SIVP.

\begin{definition}[Approximate Shortest Vector Problem ($\gamma$-SVP)]
    Given a lattice basis $B$ of an $\nd$-dimensional lattice $\LL(B)$, find a nonzero lattice vector $v \in \LL$ such that $||v|| \leq \gamma(\nd) \lambda_1(\LL)$.
\end{definition}
In particular, setting $\gamma = 1$ gives the exact Shortest Vector Problem (SVP) of finding a shortest nonzero lattice vector. Note the usage of \textit{a} shortest vector here, there may exist more than one vector $v \in \LL$ that satisfies the condition.

\begin{definition}[Decisional Approximate SVP ($\gamma$-GapSVP)]
    Given a basis $B$ of an $\nd$-dimensional lattice $\LL(B)$, where either $\lambda_1(\LL) \leq 1$ or $\lambda_1(\LL) > \gamma(n)$, determine which is the case.
\end{definition}

\begin{definition}[Approximate Shortest Independent Vectors Problem ($\gamma$-SIVP)]
    Given a full-rank basis $B$ of an $\nd$-dimensional lattice $\LL(B)$, output a set $S = \{ s_i \} \subset \LL$ of $\nd$ linearly independent lattice vectors where $|| s_i || \leq \gamma(n) \lambda_n(\LL)$ for all $i$.
\end{definition}

\begin{definition}[Bounded Distance Decoding ($\gamma$-BDD)]
    Given a basis $B$ of an $\nd$-dimensional lattice $\LL(B)$ and a target point $t \in \RR^\nd$ such that $\mathrm{dist}(t, \LL) < d = \lambda_1(\LL)/(2 \gamma(\nd))$, find the unique lattice vector closest to the target point $t$, i.e. such that $|| t - v || < d$.
\end{definition}
